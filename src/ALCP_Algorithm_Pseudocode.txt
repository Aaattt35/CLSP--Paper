Algorithm 1: Difference In Profit (Delta)
Input: cap (capacities), d (demands)

1:  procedure Delta(t, r)
2:      q_r(t)   ← Q-Operator(Cap_t, t, r)
3:      q_{r+1}(r) ← Q-Operator(Cap_r, r, r + 1)
4:      Del      ← sum_{i=r}^{q_r(t)} a_{t,i} - sum_{i=r+1}^{q_{r+1}(r)} a_{r,i}
5:      return Del
6:  end procedure

7:  procedure Q-Operator(C, t, r)
8:      set sp as initial step equal to |sum_{j=1}^T d_j|
9:      if (C - d_t - sum_{i=r}^{r+sp} d_i) < 0 then
10:         while sp ≥ 1 and (cap[t] - sum_{i=r}^{r+sp-1} d_i < 0) do
11:             sp ← sp - 1
12:         end while
13:         q ← sp + r - 1
14:     end if
15:     if (C - d_t - sum_{i=r}^{r+sp} d_i) ≥ 0 then
16:         while sp < T - sp and (cap[t] - sum_{i=r}^{r+sp+1} d_i ≥ 0) do
17:             sp ← sp + 1
18:         end while
19:         q ← sp + r
20:     end if
21:     return q
22: end procedure
